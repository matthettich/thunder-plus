# U32: stb_vorbis (the SoundFont player's decoder for SF3 samples), pinned in third_party/fetch.sh and PINS.md, as a C static library.
# Fetched alone (THUNDER_FETCH_STB_ONLY=1) when not there, so a closed build (no DSP, no pump) needs only this one file. HOST builds compile a COPY of the pin made by
# third_party/prepare-stb.sh (each patch commented there). The in-memory API only: STB_VORBIS_NO_STDIO (the core never opens a file) and STB_VORBIS_NO_PUSHDATA_API.
# Strict floating point (no fast-math, no contraction: the decoder's output must be the same on every platform up to libm).
set(TP ${CMAKE_CURRENT_SOURCE_DIR}/../third_party)
set(DEPS ${THUNDER_DEPS_DIR})
if(NOT DEPS)
  set(DEPS ${TP}/_deps)
endif()
if(NOT EXISTS ${DEPS}/stb/stb_vorbis.c)
  message(STATUS "thunder_stb_vorbis: fetching the pinned stb_vorbis.c into ${DEPS}")
  execute_process(COMMAND ${CMAKE_COMMAND} -E env THUNDER_FETCH_STB_ONLY=1 sh ${TP}/fetch.sh ${DEPS} RESULT_VARIABLE rc)
  if(NOT rc EQUAL 0)
    message(FATAL_ERROR "third_party/fetch.sh failed (needs git and network access to github.com)")
  endif()
endif()
file(SHA256 ${TP}/prepare-stb.sh stb_prep_h)
set(stb_prep_stamp "")
set(SP ${CMAKE_BINARY_DIR}/stb_prepared)
if(EXISTS ${SP}/.prepared)
  file(READ ${SP}/.prepared stb_prep_stamp)
endif()
if(NOT EXISTS ${SP}/stb_vorbis.c OR NOT stb_prep_stamp STREQUAL "${stb_prep_h} ${DEPS}")
  execute_process(COMMAND sh ${TP}/prepare-stb.sh ${DEPS}/stb ${SP} RESULT_VARIABLE rc)
  if(NOT rc EQUAL 0)
    message(FATAL_ERROR "third_party/prepare-stb.sh failed")
  endif()
  file(WRITE ${SP}/.prepared "${stb_prep_h} ${DEPS}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${TP}/prepare-stb.sh)

add_library(thunder_stb_vorbis STATIC ${SP}/stb_vorbis.c)
set_target_properties(thunder_stb_vorbis PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(thunder_stb_vorbis PRIVATE STB_VORBIS_NO_STDIO STB_VORBIS_NO_PUSHDATA_API STB_VORBIS_NO_INTEGER_CONVERSION)
if(NOT MSVC)
  target_compile_options(thunder_stb_vorbis PRIVATE -O2 -w -ffp-contract=off)
endif()
target_include_directories(thunder_stb_vorbis SYSTEM PUBLIC ${SP})
